   133→   133→   133→export function createDuel(opts: CreateDuelOpts): DuelState {
   134→   134→   134→  const rng = new Rng(opts.seed);
   135→   135→   135→  const st: DuelState = {
   136→   136→   136→    seed: opts.seed,
   137→   137→   137→    givens: opts.givens,
   138→   138→   138→    solution: opts.solution ?? null,
   139→   139→   139→    clockMs: 0,
   140→   140→   140→    players: [
   141→   141→   141→      newPlayer(0, opts.names?.[0] ?? 'You', opts.orders?.[0] ?? 'scholar'),
   142→   142→   142→      newPlayer(1, opts.names?.[1] ?? 'Foe', opts.orders?.[1] ?? 'executioner'),
   143→   143→   143→    ],
   144→   144→   144→    unitOwner: {},
   145→   145→   145→    phase: 'live',
   146→   146→   146→    winner: null,
   147→   147→   147→    winReason: null,
   148→   148→   148→    rngState: rng.state,
   149→   149→   149→    lastStatusAtMs: -CONFIG.duel.statusGlobalGapMs,
   150→   150→   150→    statusUid: 1,
   151→   151→   151→    events: [],
   152→   152→   152→    eventSeq: 1,
   153→   153→   153→    durationMs: CONFIG.duel.durationMs,
   154→   154→   154→  };
   155→   155→   155→  if (opts.magistrateSeals) {
   156→   156→   156→    for (let i = 0; i < 2; i++) {
   157→   157→   157→      const s = opts.magistrateSeals[i];
   158→   158→   158→      if (!Number.isInteger(s) || s < 1 || s > CONFIG.seals.magistrate)

   196→   196→   196→function unitComplete(st: DuelState, p: PlayerState, unit: UnitId): boolean {

   200→   200→   200→// the most nearly complete unclaimed unit on p's board (ties broken by seeded rng)
   201→   201→   201→function mostNearlyComplete(st: DuelState, p: PlayerState, r: Rng): UnitId | null {
   202→   202→   202→  const all: UnitId[] = [];
   203→   203→   203→  for (let i = 0; i < 9; i++) { all.push(`r${i}`, `c${i}`, `b${i}`); }
   204→   204→   204→  let best: UnitId[] = [];
   205→   205→   205→  let bestN = -1;
   206→   206→   206→  for (const u of all) {
   207→   207→   207→    if (st.unitOwner[u] !== undefined) continue;
   208→   208→   208→    const cells = UNIT_CELLS[u];
   209→   209→   209→    const n = cells.filter((c) => p.board[c] !== 0).length;
   210→   210→   210→    if (n === 9) continue; // must still have an empty cell to target
   211→   211→   211→    if (n > bestN) { bestN = n; best = [u]; }
   212→   212→   212→    else if (n === bestN) best.push(u);
   213→   213→   213→  }
   214→   214→   214→  if (!best.length) return null;
   215→   215→   215→  return best.length === 1 ? best[0] : r.pick(best);
   216→   216→   216→}

   225→   225→   225→export function applyStatus(st: DuelState, source: PlayerId, type: StatusType, target: PlayerId, opts: ApplyStatusOpts = {}):
   226→   226→   226→  { applied: boolean; reflected?: boolean; reason?: string } {
   227→   227→   227→  if (st.phase !== 'live') return { applied: false, reason: 'ended' };
   228→   228→   228→  // hostile-input hardening: unknown types used to push NaN-lived statuses, garbage
   229→   229→   229→  // cells/units used to be stored verbatim (adversarial suite H16/H11)
   230→   230→   230→  if (source !== 0 && source !== 1) return { applied: false, reason: 'invalid' };
   231→   231→   231→  if (target !== 0 && target !== 1) return { applied: false, reason: 'invalid' };
   232→   232→   232→  if (!VALID_STATUS.includes(type)) return { applied: false, reason: 'invalid' };
   233→   233→   233→  if (opts.cell !== undefined && (!Number.isInteger(opts.cell) || opts.cell < 0 || opts.cell > 80)) return { applied: false, reason: 'invalid' };
   234→   234→   234→  if (opts.cells && (!opts.cells.length || !opts.cells.every((c) => Number.isInteger(c) && c >= 0 && c <= 80))) return { applied: false, reason: 'invalid' };
   235→   235→   235→  if (opts.unit !== undefined && !UNIT_CELLS[opts.unit]) return { applied: false, reason: 'invalid' };
   236→   236→   236→  const dur = (CONFIG.statusDurationsMs as Record<string, number>)[type];
   237→   237→   237→  // Apothecary Distiller: statuses you apply last +2s
   238→   238→   238→  const bonus = st.players[source].order === 'apothecary' && (type === 'smudge' || type === 'miasma') ? 2000 : 0;
   239→   239→   239→  const foe = foeOf(st, source);
   240→   240→   240→  let tgt: PlayerId = target;
   241→   241→   241→
   242→   242→   242→  // Reflection (Warden Mirror) — reflect back within the armed window
   243→   243→   243→  if (foe.mirrorUntilMs > st.clockMs) {
   244→   244→   244→    foe.mirrorUntilMs = 0;
   245→   245→   245→    tgt = source;
   246→   246→   246→    pushEvent(st, 'mirrored', { player: target, abilitySource: source, status: type });
   247→   247→   247→  }
   248→   248→   248→
   249→   249→   249→  const tp = st.players[tgt];
   250→   250→   250→  // Warden Bulwark — the first incoming status each duel is negated
   251→   251→   251→  if (tp.order === 'warden' && !tp.bulwarkUsed) {
   252→   252→   252→    tp.bulwarkUsed = true;
   253→   253→   253→    pushEvent(st, 'negated', { player: tgt, status: type, by: 'bulwark' });
   254→   254→   254→    return { applied: false, reason: 'bulwark' };
   255→   255→   255→  }
   256→   256→   256→  // Ward — negate the next incoming status within 15s
   257→   257→   257→  if (tp.wardUntilMs > st.clockMs) {
   258→   258→   258→    tp.wardUntilMs = 0;
   259→   259→   259→    pushEvent(st, 'negated', { player: tgt, status: type, by: 'ward' });
   260→   260→   260→    return { applied: false, reason: 'ward' };
   261→   261→   261→  }
   262→   262→   262→  // Anti-frustration rules (spec §2)
   263→   263→   263→  if (st.clockMs > st.durationMs - CONFIG.duel.finalStatusBanMs) return { applied: false, reason: 'finalBan' };
   264→   264→   264→  if (st.clockMs - st.lastStatusAtMs < CONFIG.duel.statusGlobalGapMs) return { applied: false, reason: 'gap' };
   265→   265→   265→  if (findStatus(tp, type)) return { applied: false, reason: 'active' };
   266→   266→   266→  if ((tp.immuneUntil[type] ?? 0) > st.clockMs) return { applied: false, reason: 'immune' };
   267→   267→   267→
   268→   268→   268→  st.lastStatusAtMs = st.clockMs;
   269→   269→   269→  const inst: StatusInst = { uid: st.statusUid++, type, target: tgt, endsAtMs: st.clockMs + dur + bonus };
   270→   270→   270→  if (opts.cell !== undefined) inst.cell = opts.cell;
   271→   271→   271→  if (opts.cells) inst.cells = opts.cells;
   272→   272→   272→  if (opts.unit !== undefined) inst.unit = opts.unit;
   273→   273→   273→  tp.statuses.push(inst);
   274→   274→   274→  pushEvent(st, 'status', { player: tgt, from: source, status: type, ...opts });
   275→   275→   275→  return { applied: true };
   276→   276→   276→}

   278→   278→   278→// ---------------------------------------------------------------- placement
   279→   279→   279→export function place(st: DuelState, player: PlayerId, cell: number, digit: Digit): PlaceResult {
   280→   280→   280→  if (st.phase !== 'live') return { ok: false, reason: 'ended' };
   281→   281→   281→  // hostile-input hardening: invalid seats used to crash (TypeError on p.board), and
   282→   282→   282→  // NaN/2.5 digits passed the old range check and burned Seals as fake mistakes
   283→   283→   283→  if (player !== 0 && player !== 1) return { ok: false, reason: 'invalidTarget' };
   284→   284→   284→  const p = st.players[player];
   285→   285→   285→  if (!Number.isInteger(cell) || cell < 0 || cell > 80) return { ok: false, reason: 'invalidTarget' };
   286→   286→   286→  if (!Number.isInteger(digit) || digit < 1 || digit > 9) return { ok: false, reason: 'invalidTarget' };
   287→   287→   287→  if (p.board[cell] !== 0) return { ok: false, reason: p.board[cell] === st.givens[cell] ? 'given' : 'filled' };
   288→   288→   288→  if (findStatus(p, 'hush')) return { ok: false, reason: 'hushed' };
   289→   289→   289→  if (findStatus(p, 'chain')?.cell === cell) return { ok: false, reason: 'chained' };
   290→   290→   290→  if (!st.solution) return { ok: false, reason: 'noSolution' };
   291→   291→   291→
   292→   292→   292→  const correct = st.solution[cell] === digit;
   293→   293→   293→  if (!correct) {
   294→   294→   294→    p.mistakes++;
   295→   295→   295→    for (const u of CELL_UNITS(cell)) p.mistakesByUnit[u] = (p.mistakesByUnit[u] ?? 0) + 1;
   296→   296→   296→    const forgiven = p.order === 'scholar' && !p.marginaliaUsed;
   297→   297→   297→    if (forgiven) p.marginaliaUsed = true;
   298→   298→   298→    else p.seals = Math.max(0, p.seals - CONFIG.placement.wrongSealCost);
   299→   299→   299→    p.flinchUntilMs = st.clockMs + CONFIG.placement.flinchMs; // "Flinch"
   300→   300→   300→    pushEvent(st, 'mistake', { player, cell, digit, forgiven, sealsLeft: p.seals });
   301→   301→   301→    checkSealDeath(st, player);
   302→   302→   302→    return { ok: true, correct: false };
   303→   303→   303→  }
   304→   304→   304→
   305→   305→   305→  p.board[cell] = digit;
   306→   306→   306→  p.progress++;
   307→   307→   307→  // Momentum: each correct placement reduces all your cooldowns by 0.5s
   308→   308→   308→  for (const a of Object.values(p.abilities)) a.cdLeftMs = Math.max(0, a.cdLeftMs - CONFIG.placement.momentumReductionMs);
   309→   309→   309→  pushEvent(st, 'placed', { player, cell, digit });
   310→   310→   310→
   311→   311→   311→  // Claims
   312→   312→   312→  for (const u of CELL_UNITS(cell)) {
   313→   313→   313→    if (st.unitOwner[u] !== undefined) continue;
   314→   314→   314→    if (!unitComplete(st, p, u)) continue;
   315→   315→   315→    const foe = foeOf(st, player);
   316→   316→   316→    const quarantined = findStatus(p, 'quarantine')?.unit === u;
   317→   317→   317→    if (quarantined) {
   318→   318→   318→      // QUARANTINE: cannot be claimed by them while active — deferred until it ends
   319→   319→   319→      const q = findStatus(p, 'quarantine')!;
   320→   320→   320→      (q as StatusInst & { pendingClaim?: { player: PlayerId; unit: UnitId } }).pendingClaim = { player, unit: u };
   321→   321→   321→      pushEvent(st, 'claim', { player, unit: u, deferred: true });
   322→   322→   322→      continue;
   323→   323→   323→    }
   324→   324→   324→    resolveClaim(st, player, u);
   325→   325→   325→  }
   326→   326→   326→
   327→   327→   327→  if (p.board.every((v) => v !== 0)) endDuel(st, player, 'reckoning');
   328→   328→   328→  return { ok: true, correct: true };
   329→   329→   329→}
   330→   330→   330→
   331→   331→   331→function resolveClaim(st: DuelState, player: PlayerId, unit: UnitId) {
   332→   332→   332→  const p = st.players[player];
   333→   333→   333→  const foe = foeOf(st, player);
   334→   334→   334→  st.unitOwner[unit] = player;
   335→   335→   335→  p.claimed.push(unit);
   336→   336→   336→  let damage = CONFIG.claims.damage;
   337→   337→   337→  const clean = (p.mistakesByUnit[unit] ?? 0) === 0 && !p.augurRevealed[unit];
   338→   338→   338→  if (clean) damage += CONFIG.claims.cleanBonus;
   339→   339→   339→  if (p.reckoningUntilMs > st.clockMs) { damage += 1; p.reckoningUntilMs = 0; }
   340→   340→   340→  if (p.order === 'executioner' && foe.seals <= 3) damage += 1; // Last Rites
   341→   341→   341→  foe.seals = Math.max(0, foe.seals - damage);
   342→   342→   342→  pushEvent(st, 'claim', { player, unit, clean, damage });
   343→   343→   343→  checkSealDeath(st, foe.id);
   344→   344→   344→}

   347→   347→   347→export function useAbility(st: DuelState, player: PlayerId, abilityId: AbilityId, arg: { cell?: number; unit?: UnitId } = {}): AbilityResult {
   348→   348→   348→  if (st.phase !== 'live') return { ok: false, reason: 'ended' };
   349→   349→   349→  if (player !== 0 && player !== 1) return { ok: false, reason: 'invalidTarget' };
   350→   350→   350→  const p = st.players[player];
   351→   351→   351→  if (!ALL_ABILITIES.has(abilityId)) return { ok: false, reason: 'unknown' };
   352→   352→   352→  const rt = p.abilities[abilityId];
   353→   353→   353→  if (!rt || !ORDER_ABILITIES[p.order].includes(abilityId)) return { ok: false, reason: 'wrongOrder' };
   354→   354→   354→  if (rt.cdLeftMs > 0) return { ok: false, reason: 'cooldown' };

   357→   357→   357→  const r = rngOf(st);

   363→   363→   363→    rt.usedOnce = true;
   364→   364→   364→    if (rt.usesLeft !== null) rt.usesLeft--;

   368→   368→   368→    case 'augur': {

   378→   378→   378→      p.statuses = [];

   381→   381→   381→      pushEvent(st, 'ability', { player, ability: 'unseal', cleared });
   382→   382→   382→      setCd(abilityId);
   383→   383→   383→      break;
   384→   384→   384→    }
   385→   385→   385→    case 'fairCopy': {
   386→   386→   386→      const cell = arg.cell;
   387→   387→   387→      if (cell === undefined || !Number.isInteger(cell) || cell < 0 || cell > 80 || p.board[cell] !== 0) { commitRng(st, r); return { ok: false, reason: 'invalidTarget' }; }

   391→   391→   391→        if (p.board[c] !== 0) continue;
   392→   392→   392→        const used = new Set<number>();

   398→   398→   398→      break;

   408→   408→   408→      setCd(abilityId);
   409→   409→   409→      break;
   410→   410→   410→    }
   411→   411→   411→    case 'hush': {
   412→   412→   412→      const res = applyStatus(st, player, 'hush', foe.id);
   413→   413→   413→      result = { ok: true, applied: res.applied, statusNegated: !res.applied };
   414→   414→   414→      pushEvent(st, 'ability', { player, ability: 'hush' });
   415→   415→   415→      setCd(abilityId);

   417→   417→   417→    }

   420→   420→   420→      pushEvent(st, 'ability', { player, ability: 'reckoning', untilMs: p.reckoningUntilMs });
   421→   421→   421→      setCd(abilityId);
   422→   422→   422→      break;
   423→   423→   423→    }
   424→   424→   424→    case 'smudge': {
   425→   425→   425→      const placed: number[] = [];
   426→   426→   426→      for (let c = 0; c < 81; c++) if (foe.board[c] !== 0 && st.givens[c] === 0) placed.push(c);
   427→   427→   427→      const cells = r.shuffle(placed).slice(0, CONFIG.statusDurationsMs.smudgeDigits);
   428→   428→   428→      if (!cells.length) { commitRng(st, r); return { ok: false, reason: 'invalidTarget' }; }
   429→   429→   429→      const res = applyStatus(st, player, 'smudge', foe.id, { cells });
   430→   430→   430→      result = { ok: true, applied: res.applied, statusNegated: !res.applied };
   431→   431→   431→      pushEvent(st, 'ability', { player, ability: 'smudge', cells: res.applied ? cells : undefined });
   432→   432→   432→      setCd(abilityId);
   433→   433→   433→      break;
   434→   434→   434→    }
   435→   435→   435→    case 'tincture': {
   436→   436→   436→      if (p.seals >= CONFIG.seals.tinctureCap) { commitRng(st, r); return { ok: false, reason: 'invalidTarget' }; }
   437→   437→   437→      p.seals = Math.min(CONFIG.seals.tinctureCap, p.seals + 1);
   438→   438→   438→      pushEvent(st, 'ability', { player, ability: 'tincture', seals: p.seals });
   439→   439→   439→      setCd(abilityId);
   440→   440→   440→      break;
   441→   441→   441→    }
   442→   442→   442→    case 'miasma': {
   443→   443→   443→      const res = applyStatus(st, player, 'miasma', foe.id);
   444→   444→   444→      result = { ok: true, applied: res.applied, statusNegated: !res.applied };
   445→   445→   445→      pushEvent(st, 'ability', { player, ability: 'miasma' });
   446→   446→   446→      setCd(abilityId);
   447→   447→   447→      break;
   448→   448→   448→    }
   449→   449→   449→    case 'ward': {
   450→   450→   450→      p.wardUntilMs = st.clockMs + CONFIG.abilityCdMs.wardNegateWindowMs;
   451→   451→   451→      pushEvent(st, 'ability', { player, ability: 'ward', untilMs: p.wardUntilMs });
   452→   452→   452→      setCd(abilityId);
   453→   453→   453→      break;
   454→   454→   454→    }
   455→   455→   455→    case 'mirror': {
   456→   456→   456→      p.mirrorUntilMs = st.clockMs + CONFIG.abilityCdMs.mirrorReflectWindowMs;
   457→   457→   457→      pushEvent(st, 'ability', { player, ability: 'mirror', untilMs: p.mirrorUntilMs });
   458→   458→   458→      setCd(abilityId);
   459→   459→   459→      break;
   460→   460→   460→    }
   461→   461→   461→    case 'quarantine': {
   462→   462→   462→      const unit = arg.unit ?? mostNearlyComplete(st, foe, r) ?? undefined;
   463→   463→   463→      if (!unit || (arg.unit !== undefined && !UNIT_CELLS[arg.unit])) { commitRng(st, r); return { ok: false, reason: 'invalidTarget' }; }
   464→   464→   464→      const res = applyStatus(st, player, 'quarantine', foe.id, { unit });
   465→   465→   465→      result = { ok: true, applied: res.applied, statusNegated: !res.applied };
   466→   466→   466→      pushEvent(st, 'ability', { player, ability: 'quarantine', unit: res.applied ? unit : undefined });
   467→   467→   467→      setCd(abilityId);
   468→   468→   468→      break;
   469→   469→   469→    }
   470→   470→   470→    default:
   471→   471→   471→      commitRng(st, r);
   472→   472→   472→      return { ok: false, reason: 'unknown' };
   473→   473→   473→  }
   474→   474→   474→  commitRng(st, r);
   475→   475→   475→  return result;

   478→   478→   478→// ---------------------------------------------------------------- clock
   479→   479→   479→export function tick(st: DuelState, dtMs: number) {
   480→   480→   480→  if (st.phase !== 'live') return;
   481→   481→   481→  // hostile-input hardening: NaN used to brick the duel permanently (clock stuck at
   482→   482→   482→  // NaN, nothing ever expires), Infinity insta-ended it, negatives rewound time
   483→   483→   483→  if (!Number.isFinite(dtMs) || dtMs < 0) return;
   484→   484→   484→  st.clockMs += dtMs;
   485→   485→   485→  for (const p of st.players) {
   486→   486→   486→    // status expiry + per-type immunity
   487→   487→   487→    for (const s of [...p.statuses]) {
   488→   488→   488→      if (s.endsAtMs <= st.clockMs) {
   489→   489→   489→        p.statuses = p.statuses.filter((x) => x.uid !== s.uid);
   490→   490→   490→        p.immuneUntil[s.type] = st.clockMs + CONFIG.duel.statusImmunityMs;
   491→   491→   491→        // deferred claim under an expired quarantine resolves now
   492→   492→   492→        const pend = (s as StatusInst & { pendingClaim?: { player: PlayerId; unit: UnitId } }).pendingClaim;
   493→   493→   493→        if (pend && s.type === 'quarantine' && st.unitOwner[pend.unit] === undefined && unitComplete(st, st.players[pend.player], pend.unit)) {
   494→   494→   494→          resolveClaim(st, pend.player, pend.unit);
   495→   495→   495→        }
   496→   496→   496→        pushEvent(st, 'statusEnded', { player: p.id, status: s.type });
   497→   497→   497→      }
   498→   498→   498→    }
   499→   499→   499→    // cooldowns (Flinch pauses them)
   500→   500→   500→    if (st.clockMs >= p.flinchUntilMs) {
   501→   501→   501→      const dt = st.clockMs - p.cdSyncedAtMs;
   502→   502→   502→      if (dt > 0) for (const a of Object.values(p.abilities)) a.cdLeftMs = Math.max(0, a.cdLeftMs - dt);
   503→   503→   503→    }
   504→   504→   504→    p.cdSyncedAtMs = st.clockMs;
   505→   505→   505→  }
   506→   506→   506→  // Sudden Judgment (win condition 3)
   507→   507→   507→  if (st.clockMs >= st.durationMs) {
   508→   508→   508→    const [a, b] = st.players;
   509→   509→   509→    if (a.seals !== b.seals) endDuel(st, a.seals > b.seals ? 0 : 1, 'suddenJudgment');
   510→   510→   510→    else if (a.claimed.length !== b.claimed.length) endDuel(st, a.claimed.length > b.claimed.length ? 0 : 1, 'suddenJudgment');
   511→   511→   511→    else if (a.mistakes !== b.mistakes) endDuel(st, a.mistakes < b.mistakes ? 0 : 1, 'suddenJudgment');
   512→   512→   512→    else endDuel(st, 'draw', 'suddenJudgment');
   513→   513→   513→  }
   514→   514→   514→}
